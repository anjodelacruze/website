async function send(){
  const name=$("nm").value.trim(),ph=$("ph").value.trim();
  if(!name||ph.replace(/\D/g,"").length<7){toast("Please enter your name and a valid phone number");return}
  const del=document.querySelector('input[name="mode"]:checked').value==="Delivery";
  const addr=$("addr").value.trim(),note=$("note").value.trim();
  if(del&&addr.length<8){toast("Please enter your full delivery address");return}
  const d=new Date(),p=n=>String(n).padStart(2,"0");
  const no=`CA-${p(d.getMonth()+1)}${p(d.getDate())}-${p(d.getHours())}${p(d.getMinutes())}`;
  const when=d.toLocaleString("en-PH",{month:"short",day:"numeric",hour:"numeric",minute:"2-digit"});

  if(note)lines.push(`Notes: ${note}`);

}

}

});
